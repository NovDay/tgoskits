cmake_minimum_required(VERSION 3.20)
project(drm-smoke C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(drm-smoke src/main.c)
target_compile_options(drm-smoke PRIVATE -Wall -Wextra -Werror)
install(TARGETS drm-smoke RUNTIME DESTINATION usr/bin)
