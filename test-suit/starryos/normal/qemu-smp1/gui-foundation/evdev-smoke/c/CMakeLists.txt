cmake_minimum_required(VERSION 3.20)
project(evdev-smoke C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(evdev-smoke src/main.c)
target_compile_options(evdev-smoke PRIVATE -Wall -Wextra -Werror)
install(TARGETS evdev-smoke RUNTIME DESTINATION usr/bin)
