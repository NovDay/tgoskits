cmake_minimum_required(VERSION 3.20)
project(timerfd-smoke C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(timerfd-smoke src/main.c)
target_compile_options(timerfd-smoke PRIVATE -Wall -Wextra -Werror)
install(TARGETS timerfd-smoke RUNTIME DESTINATION usr/bin)
